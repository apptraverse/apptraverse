cmake_minimum_required(VERSION 3.25)
project(shared_chat_android LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
if(APPTRAVERSE_CMAKE_INITIAL_CACHE)
  include("${APPTRAVERSE_CMAKE_INITIAL_CACHE}")
endif()
if(APPTRAVERSE_MINIMUM_64)
  set(CMAKE_BUILD_TYPE MinSizeRel CACHE STRING "Native shipping configuration" FORCE)
endif()
get_filename_component(APPTRAVERSE_ROOT "${CMAKE_CURRENT_LIST_DIR}/../../../.." REALPATH)
if(APPTRAVERSE_MINIMUM_64 AND NOT APPTRAVERSE_CHAT_SUPPLY_HEADER)
  set(APPTRAVERSE_CHAT_SUPPLY_HEADER
      "${APPTRAVERSE_ROOT}/examples/shared_node_chat/generated/chat_supply.h"
      CACHE FILEPATH "Current embedded chat supply" FORCE)
endif()
include("${APPTRAVERSE_ROOT}/cmake/apptraverse_minimum_size.cmake")
set(APPTRAVERSE_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(APPTRAVERSE_BUILD_LEGACY_EXAMPLES OFF CACHE BOOL "" FORCE)
set(APPTRAVERSE_BUILD_AETHER_LINK ON CACHE BOOL "" FORCE)
set(APPTRAVERSE_BUILD_SHARED_NODE_CHAT ON CACHE BOOL "" FORCE)
set(BUILD_TESTING OFF CACHE BOOL "" FORCE)
add_subdirectory("${APPTRAVERSE_ROOT}" apptraverse)
add_library(shared_chat_android SHARED jni_bridge.cpp)
target_include_directories(shared_chat_android PRIVATE "${APPTRAVERSE_ROOT}/src/platform/android")
target_link_libraries(shared_chat_android PRIVATE apptraverse_shared_chat_ui apptraverse_ui_android)
