CPMAddPackage(NAME update_lab_7zip GITHUB_REPOSITORY ip7z/7zip
  GIT_TAG d3296e39e1f1f92c7b41c0c3848e65640428d42d DOWNLOAD_ONLY YES)
set(lzma_sources "${update_lab_7zip_SOURCE_DIR}/C")
add_library(update_lab_lzma1 STATIC
  "${lzma_sources}/LzmaEnc.c" "${lzma_sources}/LzFind.c"
  "${lzma_sources}/CpuArch.c" "${lzma_sources}/LzmaDec.c")
target_compile_definitions(update_lab_lzma1 PRIVATE Z7_ST)
target_include_directories(update_lab_lzma1 PUBLIC "${lzma_sources}")
add_library(update_lab_wire STATIC wire.cpp resolve.cpp state_package.cpp install.cpp)
target_link_libraries(update_lab_wire PUBLIC apptraverse PRIVATE update_lab_lzma1)
if(MSVC)
  target_compile_options(apptraverse PRIVATE /MP8)
  target_compile_options(aether PRIVATE /MP8)
elseif(CMAKE_CXX_COMPILER_ID MATCHES "Clang")
  target_compile_options(aether PUBLIC -frelaxed-template-template-args)
  target_compile_options(aether PRIVATE -fno-rtti)
  # Keep upstream SDK diagnostics visible; its optional telemetry headers emit
  # Clang warnings that are unrelated to this consumer's model policy.
  target_compile_options(aether PRIVATE -Wno-error)
endif()
target_include_directories(update_lab_wire PUBLIC ${CMAKE_CURRENT_SOURCE_DIR})
add_library(update_lab_publisher STATIC publish.cpp model.cpp supply_builder.cpp)
target_link_libraries(update_lab_publisher PUBLIC update_lab_wire)
target_compile_definitions(update_lab_publisher PUBLIC UPDATE_LAB_SCHEMA=1)
add_executable(update_lab_pack pack_main.cpp)
target_link_libraries(update_lab_pack PRIVATE update_lab_publisher)
set(seed_header "${CMAKE_CURRENT_BINARY_DIR}/seed.h")
add_custom_command(OUTPUT "${seed_header}"
  COMMAND update_lab_pack seed "${seed_header}"
  DEPENDS update_lab_pack VERBATIM)
add_library(update_lab_network STATIC network.cpp)
target_link_libraries(update_lab_network PUBLIC update_lab_wire aether)
add_executable(update_lab_provider provider_main.cpp)
target_link_libraries(update_lab_provider PRIVATE update_lab_network update_lab_publisher)
foreach(revision RANGE 0 1)
  add_executable(update_lab_e${revision} main.cpp session.cpp model.cpp download.cpp "${seed_header}")
  target_link_libraries(update_lab_e${revision} PRIVATE update_lab_network)
  target_include_directories(update_lab_e${revision} PRIVATE "${CMAKE_CURRENT_BINARY_DIR}")
  if(MSVC)
    target_compile_options(update_lab_e${revision} PRIVATE /MP4)
  endif()
  if(revision EQUAL 0)
    target_compile_definitions(update_lab_e${revision} PRIVATE UPDATE_LAB_SCHEMA=0 UPDATE_LAB_EXECUTABLE=0)
  else()
    target_compile_definitions(update_lab_e${revision} PRIVATE UPDATE_LAB_SCHEMA=1 UPDATE_LAB_EXECUTABLE=${revision})
  endif()
  if(WIN32)
    target_sources(update_lab_e${revision} PRIVATE windows.cpp)
    target_link_libraries(update_lab_e${revision} PRIVATE user32 gdi32 comctl32 comdlg32)
  else()
    target_sources(update_lab_e${revision} PRIVATE linux.cpp)
    target_link_libraries(update_lab_e${revision} PRIVATE PkgConfig::GTK3)
  endif()
endforeach()
add_executable(update_lab_e2 broken_main.cpp)
target_link_libraries(update_lab_e2 PRIVATE apptraverse_compile_policy)
add_executable(update_lab_launcher launcher.cpp)
